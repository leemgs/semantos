% Reference web links -- AAAI-27 compliance hardening.
%   AAAI-27 forbids embedded link annotations / hyperref in submissions.
%   The previous \pdfstartlink/\pdfendlink code embedded clickable
%   /Subtype /Link /URI annotations, which made it possible to submit a
%   NON-COMPLIANT PDF (and in fact one such build was produced). That code
%   has been REMOVED so that NO build of this source can ever emit a link
%   annotation, regardless of the toggle below.
%
%   Toggle behaviour (both settings emit ZERO link annotations):
%     0 = SUBMISSION/CAMERA-READY: \refurl prints nothing; reference URLs
%         are hidden. Fully AAAI-27 compliant. <-- keep this for submission.
%     1 = REVIEW-ONLY: \refurl prints each URL as PLAIN, NON-CLICKABLE text
%         (no color, no annotation) so authors can eyeball every URL. Still
%         AAAI-27 compliant, but do not submit this build (URLs add length).
%   Each .bib entry carries note={\refurl{<url>}}; the bst is unmodified.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\newcommand{\debugenablereferencelink}{0} % 0 = submission (URLs hidden, no links)
\ifnum\debugenablereferencelink=1
  \newcommand{\refurl}[1]{\newblock\url{#1}} % plain non-clickable text only
\else
  \newcommand{\refurl}[1]{} % submission: no URL, no annotation
\fi
