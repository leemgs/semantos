% Reference web-link mode. TWO output PDFs are produced from THIS source,
% selected by \debugenablereferencelink (default 0). Build BOTH with `make`
% (see the Makefile), which overrides the default on the command line via
% \def and writes distinct jobnames:
%
%   0 = SUBMISSION  ->  main.pdf
%       No URLs, no link annotations. Fully AAAI-27 compliant. UPLOAD THIS.
%   1 = DEBUG/REVIEW-ONLY  ->  main_bluelink.pdf
%       Each reference URL is shown as a BLUE, CLICKABLE link so authors can
%       click-check every reference. Implemented with the pdfTeX primitives
%       \pdfstartlink/\pdfendlink (NOT hyperref, which aaai2027.sty blocks).
%       This PDF embeds link annotations and MUST NOT be submitted to AAAI --
%       that is why it is written to a SEPARATE file (main_bluelink.pdf), so
%       the submission file main.pdf is always the compliant, link-free build.
%
%   \providecommand keeps 0 as the default for a plain `pdflatex main.tex`
%   (safe: produces the compliant main.pdf), while `make` passes
%   \def\debugenablereferencelink{1} to build main_bluelink.pdf.
%   Each .bib entry carries note={\refurl{<url>}}; the bst is unmodified.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\providecommand{\debugenablereferencelink}{0} % 0 = submission (default); 1 = blue links
\ifnum\debugenablereferencelink=1
  \usepackage{xcolor}
  \newcommand{\refurl}[1]{%
    \newblock
    \pdfstartlink
      height \ht\strutbox depth \dp\strutbox
      attr {/Border [0 0 0]}%                 hide the link border box
      user {/Subtype /Link
            /A << /S /URI /URI (#1) >>}%      clickable URI annotation
    \textcolor{blue}{\urlstyle{rm}\url{#1}}%   blue URL text
    \pdfendlink}
\else
  \newcommand{\refurl}[1]{}%                  submission: no URL, no annotation
\fi
