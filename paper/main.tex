%       resulting PDF embeds link annotations and MUST NOT be submitted --
%       it is only for the author to click-check every reference URL.
%   0 = SUBMISSION/CAMERA-READY: no colors, no links, URLs hidden. Fully
%       AAAI-27 compliant (no hyperref, no link annotations).

  \usepackage{xcolor}
  \newcommand{\refurl}[1]{%
    \newblock
    \pdfstartlink
      height \ht\strutbox depth \dp\strutbox
      attr {/Border [0 0 0]}%                 hide the link border box
      user {/Subtype /Link
            /A << /S /URI /URI (#1) >>}%      clickable URI annotation
    \textcolor{blue}{\urlstyle{rm}\url{#1}}%   blue URL text
    \pdfendlink}
\else
  \newcommand{\refurl}[1]{}
